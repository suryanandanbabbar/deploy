# ── Frontend serving ────────────────────────────────────────────────
# Production frontend is built into frontend/dist.
PROJECT_ROOT = Path(__file__).resolve().parents[2]
FRONTEND_DIST = PROJECT_ROOT / "frontend" / "dist"

if not FRONTEND_DIST.exists():
    logger.warning("Frontend dist directory not found: %s", FRONTEND_DIST)

@app.get("/", include_in_schema=False)
async def serve_frontend():
    index_file = FRONTEND_DIST / "index.html"

    if not index_file.exists():
        return JSONResponse(
            status_code=503,
            content={
                "error": "Frontend build not found.",
                "expected_path": str(index_file),
            },
        )

    return FileResponse(index_file)


@app.get("/{full_path:path}", include_in_schema=False)
async def frontend_spa_fallback(full_path: str):
    """
    Serve React SPA routes while leaving /api/* routes to FastAPI routers.
    """
    requested_file = FRONTEND_DIST / full_path

    # Serve an existing static file if requested.
    if requested_file.is_file():
        return FileResponse(requested_file)

    # React client-side route → index.html.
    index_file = FRONTEND_DIST / "index.html"

    if not index_file.exists():
        return JSONResponse(
            status_code=503,
            content={"error": "Frontend build not found."},
        )

    return FileResponse(index_file)
