%sh

cd /tmp/protected-app-test

VENV=/tmp/nuitka311

PYTHONPATH=/tmp/protected-app-test \
"$VENV/bin/python" -m uvicorn backend.app.main:app \
    --host 127.0.0.1 \
    --port 8000 \
    > /tmp/protected-uvicorn.log 2>&1 &

SERVER_PID=$!

sleep 5

echo "=== SERVER LOG ==="
cat /tmp/protected-uvicorn.log

echo
echo "=== API CONFIG ==="
curl -i http://127.0.0.1:8000/api/config

STATUS=$?

echo
echo "=== STOPPING SERVER ==="
kill $SERVER_PID 2>/dev/null || true
wait $SERVER_PID 2>/dev/null || true

exit $STATUS
