# ============================================================
# 3. Compile one NORMAL Python module
#
# IMPORTANT:
# __init__.py is deliberately excluded.
# Nuitka --mode=module cannot compile __init__.py directly.
# ============================================================

compile_module() {

    SRC="$1"

    REL="${SRC#$PROJECT/}"
    REL_NO_EXT="${REL%.py}"

    MODULE_NAME="$(basename "$SRC" .py)"
    REL_DIR="$(dirname "$REL")"

    DEST_DIR="$FINAL_ROOT/$REL_DIR"

    mkdir -p "$DEST_DIR"

    TMP_DIR="$BUILD_ROOT/${MODULE_NAME}_$$"

    rm -rf "$TMP_DIR"
    mkdir -p "$TMP_DIR"

    echo ""
    echo "------------------------------------------------------------"
    echo "Compiling"
    echo "  Source : $REL"
    echo "  Module : $MODULE_NAME"
    echo "------------------------------------------------------------"

    (
        cd "$PROJECT"

        "$VENV/bin/python" -m nuitka \
            --mode=module \
            --output-dir="$TMP_DIR" \
            --remove-output \
            "$SRC"
    )

    SO_FILE="$(find "$TMP_DIR" \
        -maxdepth 1 \
        -type f \
        -name "${MODULE_NAME}.cpython-311-*.so" \
        -print -quit)"

    if [ -z "$SO_FILE" ]; then

        echo ""
        echo "============================================================"
        echo "ERROR: Nuitka did not produce the expected .so"
        echo "============================================================"
        echo "Source:"
        echo "$REL"
        echo ""
        echo "Generated files:"
        find "$TMP_DIR" -maxdepth 2 -type f -print
        exit 1
    fi

    cp "$SO_FILE" "$DEST_DIR/"

    echo "Created:"
    echo "  $DEST_DIR/$(basename "$SO_FILE")"

    rm -rf "$TMP_DIR"
}

# ============================================================
# 4. Create package marker
#
# Python requires __init__.py for the package hierarchy.
#
# These are generated package markers, NOT the original source.
# ============================================================

create_package_marker() {

    PACKAGE_DIR="$1"

    mkdir -p "$PACKAGE_DIR"

    cat > "$PACKAGE_DIR/__init__.py" <<'EOF'
# Protected package marker.
EOF
}
