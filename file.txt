#!/usr/bin/env bash

echo "========================================"

echo "HOST: $HOST"
echo "PORT: $PORT"
echo "========================================"

    --host "$HOST" \
    --port "$PORT"
EOF

chmod +x "$FINAL/start.sh"

echo "========================================"
echo " FINAL ARTIFACT CREATED"
echo "========================================"

find "$FINAL" -type f -printf '%p\n' | sort

%sh

PROJECT="/Workspace/Users/in_suryanandanb@exlservice.com/Alteryx-nuitka-test"
FINAL="$PROJECT/deployment/final-databricks-app"

cd "$PROJECT"

echo "========================================"
echo " FILE AUDIT"
echo "========================================"

echo
echo "=== ALL FILES ==="
find "$FINAL" -type f -printf '%p\n' | sort

echo
echo "=== PYTHON SOURCE FILES ==="

PY_FILES=$(find "$FINAL" -type f \
    \( -name "*.py" -o -name "*.pyc" \) \
    ! -path "$FINAL/backend/__init__.py" \
    -print)

if [ -n "$PY_FILES" ]; then
    echo "ERROR: PROPRIETARY PYTHON FILES FOUND"
    echo "$PY_FILES"
    exit 1
else
    echo "PASS: No proprietary .py/.pyc files"
fi

echo
echo "=== PYTHON CACHE ==="

CACHE=$(find "$FINAL" -type d -name "__pycache__" -print)

if [ -n "$CACHE" ]; then
    echo "ERROR: __pycache__ FOUND"
    echo "$CACHE"
    exit 1
else
    echo "PASS: No __pycache__"
fi

echo
echo "=== ENVIRONMENT FILES ==="

ENV_FILES=$(find "$FINAL" -type f \
    \( -name ".env" -o -name ".env.*" \) \
    -print)

if [ -n "$ENV_FILES" ]; then
    echo "ERROR: ENV FILE FOUND"
    echo "$ENV_FILES"
    exit 1
else
    echo "PASS: No .env files"
fi

echo
echo "=== FILE SIZE AUDIT ==="

find "$FINAL" -type f -printf '%s %p\n' \
    | sort -nr \
    | head -30

echo
echo "=== FILES >= 10 MB ==="

LARGE=$(find "$FINAL" -type f -size +10M -print)

if [ -n "$LARGE" ]; then
    echo "ERROR: FILE(S) EXCEED DATABRICKS 10 MB LIMIT"
    echo "$LARGE"
    exit 1
else
    echo "PASS: No file exceeds 10 MB"
fi

echo
echo "========================================"
echo " SECURITY / SIZE AUDIT: PASS"
echo "========================================"
