echo "===== TESTING BACKEND.AWA NAMESPACE ====="

mkdir -p deployment/namespace_test/backend/awa/expressions

echo
echo "--- COPYING BACKEND MARKER ---"

touch deployment/namespace_test/backend/__init__.py

echo

cp \
  deployment/backend_awa_package_py311/awa.cpython-311-x86_64-linux-gnu.so \
  deployment/namespace_test/backend/

cp \
  deployment/backend_awa_package_py311/awa.pyi \
  deployment/namespace_test/backend/

cp \
  deployment/app_package_py311/app.cpython-311-x86_64-linux-gnu.so \
  deployment/namespace_test/backend/

  deployment/app_package_py311/app.pyi \
  deployment/namespace_test/backend/

cp \
  deployment/grammar.lark \
  deployment/namespace_test/backend/awa/expressions/grammar.lark

find deployment/namespace_test -type f | sort

    exit 1
fi

echo "NO AWA SOURCE PACKAGE MARKERS: PASS"

cd deployment/namespace_test
