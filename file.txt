%sh
set -e

OUT="$PWD/deployment/protected_v2"

echo "=============================================="
echo " DATABRICKS STARTUP CONFIGURATION VALIDATION"
echo "=============================================="

echo
echo "=== 1. app.yaml ==="

cat "$OUT/app.yaml"

echo
echo "=== 2. VERIFY app.yaml START COMMAND ==="

grep -q '"/bin/bash"' "$OUT/app.yaml"
grep -q '"./start.sh"' "$OUT/app.yaml"

echo "app.yaml command: PASS"

echo
echo "=== 3. start.sh ==="

cat "$OUT/start.sh"

echo
echo "=== 4. SHELL SYNTAX ==="

bash -n "$OUT/start.sh"

echo "start.sh syntax: PASS"

echo
echo "=== 5. EXECUTE PERMISSION ==="

if [ ! -x "$OUT/start.sh" ]; then
    echo "ERROR: start.sh is not executable"
    chmod +x "$OUT/start.sh"
fi

ls -l "$OUT/start.sh"

echo
echo "=== 6. DATABRICKS PORT HANDLING ==="

grep -q 'DATABRICKS_APP_PORT' "$OUT/start.sh"

echo "DATABRICKS_APP_PORT handling: PASS"

echo
echo "=== 7. UVICORN ENTRYPOINT ==="

grep -q 'backend.app.main:app' "$OUT/start.sh"

echo "backend.app.main:app entrypoint: PASS"

echo
echo "=============================================="
echo " STARTUP CONFIGURATION: PASS"
echo "=============================================="
