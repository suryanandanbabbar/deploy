%sh
set -e

ROOT="$PWD"
OUT="$ROOT/deployment/protected_v2"

echo "===== FIXING FRONTEND DIST LAYOUT ====="

echo
echo "--- REMOVE INCORRECT FRONTEND CONTENT ---"

rm -rf "$OUT/frontend"

echo
echo "--- CREATE CORRECT DIRECTORY ---"

mkdir -p "$OUT/frontend"

echo
echo "--- COPY FRONTEND DIST ---"

cp -R \
  "$ROOT/frontend/dist" \
  "$OUT/frontend/dist"

echo
echo "--- VERIFY ---"

find "$OUT/frontend" -type f -printf '%p\n' | sort

echo
echo "--- CHECK REQUIRED PATH ---"

if [ ! -d "$OUT/frontend/dist" ]; then
    echo "ERROR: frontend/dist still missing"
    exit 1
fi

if [ ! -f "$OUT/frontend/dist/index.html" ]; then
    echo "ERROR: frontend/dist/index.html missing"
    exit 1
fi

echo "frontend/dist: PASS"

echo
echo "===== FRONTEND LAYOUT FIX COMPLETE ====="
