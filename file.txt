%sh

cd /tmp/protected-full-test

VENV=/tmp/nuitka311

PYTHONPATH=/tmp/protected-full-test \
"$VENV/bin/python" -m uvicorn backend.app.main:app \
    --host 127.0.0.1 \
    --port 8000 \
    > /tmp/protected-full-server.log 2>&1 &

SERVER_PID=$!

sleep 4

echo "=== ROOT ==="
curl -i http://127.0.0.1:8000/

echo
echo "=== INDEX.HTML DIRECT ==="
curl -i http://127.0.0.1:8000/index.html

echo
echo "=== SERVER LOG AFTER REQUESTS ==="
cat /tmp/protected-full-server.log

echo
echo "=== STOPPING ==="
kill "$SERVER_PID" 2>/dev/null || true
wait "$SERVER_PID" 2>/dev/null || true
