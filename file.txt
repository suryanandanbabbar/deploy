%sh
set -e

echo "===== PRESERVING AWA RUNTIME GRAMMAR ====="

GRAMMAR="backend/awa/expressions/grammar.lark"
TARGET="deployment/grammar.lark"

echo
echo "--- SOURCE GRAMMAR ---"

echo "--- VERIFYING TARGET ---"
ls -lh "$TARGET"

echo
echo "--- SHA256 CHECK ---"
sha256sum "$GRAMMAR"
sha256sum "$TARGET"
